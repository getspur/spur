# jute-deck Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a notebook-native present mode to `jute-notebook` where each cell renders 1:1 as a slide, with an optional Spur-worker-driven agent layer for drafting/restructuring/polishing decks. v1 is **view-only** — no export.

**Architecture:** Three layers added to the existing `crates/spur-notebook` crate. (1) Cell metadata gains a typed `jute_deck` namespace alongside the existing `spur` namespace. (2) A pure `cellToSlide` transform feeds layout components rendered on a new `/present` wouter route inside the same Tauri window. (3) Four AI commands in the existing `NotebookCommandMenu` dispatch through `mcp__spur-mcp__delegate_to_worker` with a `mcp__notebook__*`-only tool allowlist.

**Tech Stack:** Rust (notebook MCP, daemon notebook store, Tauri backend, ts-rs codegen), TypeScript + React 19 + wouter + cmdk + zustand (jute frontend), vitest (new, for layout-inference unit tests).

**Spec reference:** `docs/superpowers/specs/2026-05-26-jute-deck-design.md`

```rust
Op::SetJuteDeckMetadata { id, patch, expected_version } => {
    let idx = find_cell_index(root, &id)
        .ok_or_else(|| OpError::CellNotFound(id.clone()))?;
    let cell = &mut root.cells[idx];
    let current_version = cell_version(cell).unwrap_or(0);
    if current_version != expected_version {
        return Err(OpError::StaleVersion { expected: expected_version, actual: current_version });
    }
    let metadata = cell_metadata_mut(cell);
    let mut merged = metadata.jute_deck.clone().unwrap_or_default();
    if patch.layout.is_some() { merged.layout = patch.layout; }
    if patch.hidden.is_some() { merged.hidden = patch.hidden; }
    if patch.speaker_notes.is_some() { merged.speaker_notes = patch.speaker_notes.clone(); }
    if patch.theme_override.is_some() { merged.theme_override = patch.theme_override.clone(); }
    if patch.fragments.is_some() { merged.fragments = patch.fragments; }
    if patch.background.is_some() { merged.background = patch.background.clone(); }
    metadata.jute_deck = Some(merged);
    let kind = DeltaKind::CellMetadataUpdated;
    let metadata_update = Some((id.clone(), Some("brain".to_string())));
    (kind, metadata_update)
}
```

```ts
function setCellMetadata(
  notebook: Notebook,
  params: AgentSetCellMetadata,
): { ok: true; version: number } {
  const { id, patch, expected_version } = params;
  if (!id) {
    throw new AgentHandlerError(
      "invalid_params",
      "notebook.set_cell_metadata requires { id, patch, expected_version }",
    );
  }
  if (expected_version < 1) {
    throw new AgentHandlerError(
      "invalid_params",
      "notebook.set_cell_metadata expected_version must be >= 1",
    );
  }

  // Atomic check + mutation: NO await between these two lines.
  const cell = notebook.getCellSnapshotById(id);
  if (!cell) {
    throw new AgentHandlerError("cell_not_found", `cell ${id} not found`);
  }
  const currentVersion = cell.metadata?.spur?.version ?? 0;
  if (currentVersion !== expected_version) {
    throw new AgentHandlerError(
      "stale_version",
      `expected version ${expected_version}, actual ${currentVersion}`,
    );
  }
  const nextVersion = notebook.mergeCellJuteDeckMetadata(id, patch);
  return { ok: true, version: nextVersion };
}
```

This assumes two methods on the `Notebook` zustand store: `getCellSnapshotById(id)` and `mergeCellJuteDeckMetadata(id, patch)`.

```ts
mergeCellJuteDeckMetadata(id: string, patch: Partial<JuteDeckCellMetadata>): number {
  const next = this.store.getState();  // for type inference; use the real pattern
  let newVersion = 0;
  this.store.setState((state) => {
    const cell = state.cells[id];
    if (!cell) return state;
    const existing = cell.metadata?.jute_deck ?? {};
    const merged: JuteDeckCellMetadata = {
      ...existing,
      ...Object.fromEntries(
        Object.entries(patch).filter(([, v]) => v !== undefined),
      ),

    newVersion = spur.version + 1;
    return {
      ...state,
      cells: {
        ...state.cells,
        [id]: {
          ...cell,
          metadata: {
            ...cell.metadata,
            jute_deck: merged,
            spur: { ...spur, version: newVersion, last_edited_by: "brain" },
          },
        },
      },
    };
  });
  return newVersion;
}
```

```ts
import type { Cell } from "@/bindings/Cell";
import type { CellMetadata } from "@/bindings/CellMetadata";
import type { JuteDeckNotebookMetadata } from "@/bindings/JuteDeckNotebookMetadata";

import type { Block, ResolvedLayout, SlideSpec } from "./types";

const H1_ONLY = /^\s*#\s+[^\n]+(\n[^\n#]+)?\s*$/;
const H2_ONLY = /^\s*##\s+[^\n]+\s*$/;
const HAS_BULLETS = /(^|\n)\s*[-*]\s+/;
const HAS_H1 = /(^|\n)\s*#\s+/;

export function cellToSlide(
  cell: Cell,
  deck: JuteDeckNotebookMetadata | undefined,
): SlideSpec | null {
  const meta: CellMetadata = cell.metadata ?? {};
  const jd = meta.jute_deck;

  if (jd?.hidden === true) return null;

  const explicit = jd?.layout && jd.layout !== "auto" ? jd.layout : undefined;
  const layout: ResolvedLayout = explicit ?? inferLayout(cell);

  const source = typeof cell.source === "string" ? cell.source : (cell.source ?? []).join("");
  const blocks = buildBlocks(cell, source, layout);

  const theme = jd?.theme_override ?? deck?.theme ?? "minimal-light";

  return {
    id: cell.id,
    layout,
    blocks,
    speakerNotes: jd?.speaker_notes,
    theme,
    background: jd?.background,
    fragments: jd?.fragments ?? false,
  };
}

function inferLayout(cell: Cell): ResolvedLayout {
  if (cell.type === "code") return "output";
  if (cell.type === "raw" || cell.type === "html") return "blank";

  const source = typeof cell.source === "string" ? cell.source : (cell.source ?? []).join("");

  // First-matching-rule-wins; order is intentional.
  if (H1_ONLY.test(source.trim())) return "title";
  if (H2_ONLY.test(source.trim())) return "section";
  // H1 + anything else still wins as "title" before bullets get checked,
  // because a deck-author who started with # almost certainly meant a title.
  if (HAS_H1.test(source)) return "title";
  if (HAS_BULLETS.test(source)) return "bullets";
  return "content";
}

function buildBlocks(cell: Cell, source: string, layout: ResolvedLayout): Block[] {
  if (cell.type === "code") {
    if (layout === "code") return [{ kind: "code", lang: detectLang(cell), source }];
    if (layout === "code-output")
      return [
        { kind: "code", lang: detectLang(cell), source },
        { kind: "output", outputs: cell.outputs ?? [] },
      ];
    return [{ kind: "output", outputs: cell.outputs ?? [] }];
  }
  // markdown / raw / html
  return [{ kind: "markdown", md: source }];
}

function detectLang(cell: Cell): string {
  // Try metadata first, fall back to "python".
  const lang = (cell.metadata as any)?.kernel_language ?? "python";
  return lang;
}
```

```tsx
import clsx from "clsx";

import type { Block } from "../types";
import { resolveTheme } from "../themes";

type Props = { blocks: Block[]; themeId: string; fragmentIndex?: number };

export default function BulletsLayout({ blocks, themeId, fragmentIndex = 0 }: Props) {
  const theme = resolveTheme(themeId);
  // Extract first heading + first bullet block from the markdown source. The
  // markdown block is the whole cell source; parse it shallowly.
  const md = (blocks.find((b) => b.kind === "markdown") as { kind: "markdown"; md: string } | undefined)?.md ?? "";
  const lines = md.split("\n");
  const headingLine = lines.find((l) => /^#{1,3}\s+/.test(l)) ?? "";
  const heading = headingLine.replace(/^#{1,3}\s+/, "");
  const bullets = lines.filter((l) => /^\s*[-*]\s+/.test(l)).map((l) => l.replace(/^\s*[-*]\s+/, ""));
  const hasFragments = bullets.length > 1 && fragmentIndex > 0;

  return (
    <div className="flex h-full flex-col justify-center">
      {heading && (
        <h2 className={clsx("mb-6 text-[3cqi] font-bold", theme.heading)}>{heading}</h2>
      )}
      <ul className="list-none space-y-3 p-0">
        {bullets.map((b, i) => (
          <li
            key={i}
            className={clsx(
              "flex items-start text-[2cqi] leading-snug transition-opacity",
              theme.body,
              hasFragments && i > fragmentIndex && "opacity-30",
            )}
          >
            <span className={clsx("mr-3 font-bold", theme.accent)}>→</span>
            <span dangerouslySetInnerHTML={{ __html: renderInlineMarkdown(b) }} />
          </li>
        ))}
      </ul>
    </div>
  );
}

function renderInlineMarkdown(s: string): string {
  // Minimal inline rendering: **bold**, *italic*. Escape HTML first.
  const escaped = s
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
  return escaped
    .replace(/\*\*([^*]+)\*\*/g, "<b>$1</b>")
    .replace(/\*([^*]+)\*/g, "<i>$1</i>");
}
```

```tsx
import { useEffect, useMemo, useState } from "react";
import { useLocation, useSearch } from "wouter";
import { useStore } from "zustand";

import { Notebook, NotebookContext } from "@/stores/notebook";
import PresentChrome from "@/ui/deck/PresentChrome";
import SpeakerNotes from "@/ui/deck/SpeakerNotes";
import SlideFrame from "@/ui/deck/SlideFrame";
import { LayoutFor } from "@/ui/deck/layouts";
import { cellToSlide } from "@/ui/deck/cellToSlide";
import type { SlideSpec } from "@/ui/deck/types";

export default function PresentPage() {
  const [, setLocation] = useLocation();
  const { path } = Object.fromEntries(new URLSearchParams(useSearch()));
  const notebook = useMemo(() => new Notebook(), []);

  useEffect(() => {
    if (path) void notebook.loadNotebookFromPath(path);
  }, [notebook, path]);

  const cells = useStore(notebook.store, (s) => s.cellOrder.map((id) => s.cells[id]));
  const deckMeta = useStore(notebook.store, (s) => s.notebookMetadata?.jute_deck);

  const slides: SlideSpec[] = useMemo(
    () =>
      cells
        .map((c) => cellToSlide(c, deckMeta))
        .filter((s): s is SlideSpec => s !== null),
    [cells, deckMeta],
  );

  const [idx, setIdx] = useState(0);
  const [fragmentIndex, setFragmentIndex] = useState(0);
  const [notesOpen, setNotesOpen] = useState(false);
  const [blackout, setBlackout] = useState(false);

  useEffect(() => setFragmentIndex(0), [idx]);

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      const slide = slides[idx];
      const maxFragments =
        slide?.layout === "bullets" && slide.fragments
          ? (slide.blocks.find((b) => b.kind === "bullets") as any)?.items.length ?? 1
          : 1;

      switch (e.key) {
        case "ArrowRight":
        case " ":
        case "PageDown":
          e.preventDefault();
          if (slide?.fragments && fragmentIndex < maxFragments - 1) {
            setFragmentIndex((f) => f + 1);
          } else if (idx < slides.length - 1) {
            setIdx((i) => i + 1);
          }
          break;
        case "ArrowLeft":
        case "PageUp":
          e.preventDefault();
          if (fragmentIndex > 0) setFragmentIndex((f) => f - 1);
          else if (idx > 0) setIdx((i) => i - 1);
          break;
        case "Home":
          setIdx(0);
          break;
        case "End":
          setIdx(slides.length - 1);
          break;
        case "s":
        case "S":
          setNotesOpen((v) => !v);
          break;
        case "b":
        case "B":
          setBlackout((v) => !v);
          break;
        case "Escape":
          setLocation(path ? `/notebook?path=${encodeURIComponent(path)}` : "/");
          break;
      }
    }
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [slides, idx, fragmentIndex, path, setLocation]);

  const slide = slides[idx];

  return (
    <NotebookContext.Provider value={notebook}>
      <main className="h-screen w-screen overflow-hidden bg-black">
        {blackout || !slide ? (
          <div className="h-full w-full bg-black" />
        ) : (
          <SlideFrame themeId={slide.theme} background={slide.background}>
            <LayoutFor slide={slide} fragmentIndex={fragmentIndex} />
          </SlideFrame>
        )}
        <PresentChrome current={idx} total={slides.length} />
        {notesOpen && <SpeakerNotes notes={slide?.speakerNotes} />}
      </main>
    </NotebookContext.Provider>
  );
}
```

- [ ] **Step 2: Create `dispatch.ts`**

```ts
import { invoke } from "@tauri-apps/api/core";

import type { Notebook } from "@/stores/notebook";

import { PROMPTS } from "./prompts";

type Kind = keyof typeof PROMPTS;

export async function dispatchDeckCommand(
  notebook: Notebook,
  kind: Kind,
  userPrompt: string,
): Promise<{ delegation_id: string }> {
  const summary = summarizeNotebook(notebook);
  const task = `${PROMPTS[kind]}\n\nUser's request: ${userPrompt}\n\nCurrent notebook (${summary.cells.length} cells):\n${JSON.stringify(summary, null, 2)}`;

  return await invoke<{ delegation_id: string }>("spur_delegate_to_worker", {
    task,
    workerType: "coder",
    toolAllowlist: ["mcp__notebook__*"],
  });
}

function summarizeNotebook(notebook: Notebook) {
  const state = notebook.store.getState();
  const cells = (state.cellOrder ?? Object.keys(state.cells)).map((id: string) => {
    const cell = state.cells[id];
    const source = typeof cell.source === "string" ? cell.source : (cell.source ?? []).join("");
    return {
      id,
      type: cell.type,
      layout: cell.metadata?.jute_deck?.layout ?? "auto",
      preview: source.slice(0, 80),
    };
  });
  return { path: state.path, cells };
}
```
