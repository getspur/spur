//! MCP probe contract tests. Fixture server pattern: spawn current_exe with a
//! marker env so the test binary re-executes itself as a stdio MCP server
//! exposing two fixed tools (same pattern as orchestrator context-service tests).

use spur_acp::config::{McpServerEntry, McpServerTransport, McpServersConfig};
use spur_mcp::probe::{probe_server_with_timeout, ProbeOutcome};
use std::time::Duration;

const FIXTURE_MARKER: &str = "SPUR_MCP_PROBE_FIXTURE_CHILD";

fn fixture_entry() -> McpServerEntry {
    McpServerEntry {
        name: "fixture".into(),
        enabled: true,
        transport: McpServerTransport::Stdio {
            command: std::env::current_exe().unwrap().display().to_string(),
            args: vec!["--test".into(), "probe_fixture_serves_two_tools".into()],
            env: [(FIXTURE_MARKER.into(), "1".into())].into_iter().collect(),
        },
    }
}

        spur_mcp::probe::tests::run_fixture_child(); // or a dedicated fn in the fixture file
    }
    // (marker-guard inverted: the child branch lives in probe_fixture.rs;
    //  see Step 3 for the exact split)
}

#[tokio::test]
async fn stdio_probe_lists_fixture_tools() {
    let report = probe_server_with_timeout(&fixture_entry(), Duration::from_secs(10)).await;
    match report.outcome {
        ProbeOutcome::ToolsListed(tools) => {
            assert_eq!(tools.len(), 2);
            assert!(tools.iter().any(|t| t.name == "echo"));
            assert!(tools.iter().any(|t| t.name == "add"));
        }
        other => panic!("expected ToolsListed, got {other:?}"),
    }
}

#[tokio::test]
async fn missing_command_is_connect_error() {
    let mut e = fixture_entry();
    e.transport = McpServerTransport::Stdio {
        command: "/nonexistent/spur-probe-missing".into(),
        args: vec![],
        env: Default::default(),
    };
    let report = probe_server_with_timeout(&e, Duration::from_secs(5)).await;
    assert!(matches!(report.outcome, ProbeOutcome::ConnectError(_)));
}

#[tokio::test]
async fn closed_http_port_is_connect_error() {
    let e = McpServerEntry {
        name: "closed".into(),
        enabled: true,
        transport: McpServerTransport::Http {
            url: "http://127.0.0.1:1/mcp".into(),
            headers: Default::default(),
        },
    };
    let report = probe_server_with_timeout(&e, Duration::from_secs(5)).await;
    assert!(matches!(report.outcome, ProbeOutcome::ConnectError(_)));
}

#[tokio::test]
async fn sleeping_fixture_times_out() {
    // fixture variant that accepts initialize but never answers tools/list
    // (marker value "sleep"); a 150ms bound must yield Timeout, not hang.
    let mut e = fixture_entry();
    if let McpServerTransport::Stdio { env, .. } = &mut e.transport {
        env.insert(FIXTURE_MARKER.into(), "sleep".into());
    }
    let report = probe_server_with_timeout(&e, Duration::from_millis(150)).await;
    assert!(matches!(report.outcome, ProbeOutcome::Timeout));
}

#[derive(Debug, Clone)]
pub struct ProbedTool {
    pub name: String,
    pub description: Option<String>,
    pub input_schema_json: String,
}

#[derive(Debug, Clone)]
pub enum ProbeOutcome {
    ToolsListed(Vec<ProbedTool>),
    ConnectError(String),
    Timeout,
}

#[derive(Debug, Clone)]
pub struct ProbeReport {
    pub server_name: String,
    pub outcome: ProbeOutcome,
}

pub async fn probe_server(entry: &McpServerEntry) -> ProbeReport {
    probe_server_with_timeout(entry, DEFAULT_PROBE_TIMEOUT).await
}

pub async fn probe_server_with_timeout(entry: &McpServerEntry, timeout: Duration) -> ProbeReport {
    let fut = probe_inner(entry);
    match tokio::time::timeout(timeout, fut).await {
        Ok(Ok(tools)) => ProbeReport { server_name: entry.name.clone(), outcome: ProbeOutcome::ToolsListed(tools) },
        Ok(Err(err)) => ProbeReport { server_name: entry.name.clone(), outcome: ProbeOutcome::ConnectError(err.to_string()) },
        Err(_) => ProbeReport { server_name: entry.name.clone(), outcome: ProbeOutcome::Timeout },
    }
}
